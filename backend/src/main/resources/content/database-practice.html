CREATE DATABASE IF NOT EXISTS study_app DEFAULT CHARACTER SET utf8mb4;
USE study_app;

-- 계정 테이블
CREATE TABLE IF NOT EXISTS users (
    id          INT AUTO_INCREMENT PRIMARY KEY,
    username    VARCHAR(50)  NOT NULL UNIQUE,
    password    VARCHAR(255) NOT NULL,            -- SHA-256 해시
    role        ENUM('admin', 'guest') NOT NULL DEFAULT 'guest',
    is_active   TINYINT(1)   NOT NULL DEFAULT 1,  -- 1:활성, 0:비활성
    created_at  DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at  DATETIME     NULL     DEFAULT  NULL
);

-- 공부 기록 테이블
CREATE TABLE IF NOT EXISTS study_logs (
    id          INT AUTO_INCREMENT PRIMARY KEY,
    user_id     INT          NOT NULL,
    title       VARCHAR(200) NOT NULL,
    content     TEXT         NOT NULL,
    study_date  DATE         NOT NULL DEFAULT (CURRENT_DATE),
    created_at  DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at  DATETIME     NULL     DEFAULT  NULL,
    FOREIGN KEY (user_id) REFERENCES users(id)
);

-- 초기 admin 계정 (비밀번호: admin123 의 SHA-256)
INSERT IGNORE INTO users (username, password, role)
VALUES (
    'admin',
    SHA2('admin123', 256),
    'admin'

      import os
import hashlib
import getpass
from dotenv import load_dotenv
import pymysql

# ===== DB 설정 =====
# .env 파일에서 DB 접속 정보를 읽어 딕셔너리로 반환
def load_config():
    load_dotenv()
    return {
        "host": os.getenv("DB_HOST"),
        "port": int(os.getenv("DB_PORT", 3306)),
        "user": os.getenv("DB_USER"),
        "password": os.getenv("DB_PASSWORD"),
        "database": os.getenv("DB_NAME"),
        "charset": os.getenv("DB_CHARSET", "utf8mb4"),
    }

DB_CONFIG = load_config()

# ===== 공통 함수 =====
# DB에 연결해 connection 객체 반환
def get_conn():
    return pymysql.connect(**DB_CONFIG)

# 비밀번호를 SHA-256으로 해시
def hash_pw(password):
    return hashlib.sha256(password.encode()).hexdigest()

# SQL 실행 공통 헬퍼 (fetch=True면 조회 결과 반환)
def run(sql, params=None, fetch=False):
    conn = get_conn()
    try:
        with conn.cursor(pymysql.cursors.DictCursor) as cur:
            cur.execute(sql, params or ())
            if fetch:
                return cur.fetchall()
            conn.commit()
    finally:
        conn.close()

# ===== 로그인 =====
# 아이디/비밀번호 인증 후 사용자 정보 반환 (실패 시 None)
def login():
    username = input("아이디: ").strip()
    password = getpass.getpass("비밀번호: ")
    rows = run(
        "SELECT * FROM users WHERE username=%s AND password=%s AND is_active=1",
        (username, hash_pw(password)),
        fetch=True,
    )
    if rows:
        print(f"\n[{rows[0]['role']}] {username} 님 로그인 성공\n")
        return rows[0]
    print("로그인 실패 (계정·비밀번호 확인 또는 비활성 계정)\n")
    return None

# ===== 공부 기록 기능 =====
# 공부 기록 목록 조회
def list_logs():
    rows = run(
        "SELECT l.id, u.username, l.title, l.study_date "
        "FROM study_logs l JOIN users u ON l.user_id=u.id ORDER BY l.id DESC",
        fetch=True,
    )
    print("\n--- 공부 기록 목록 ---")
    for r in rows:
        print(f"[{r['id']}] {r['study_date']} | {r['title']} ({r['username']})")
    print()

# 공부 기록 상세 조회
def view_log():
    log_id = input("조회할 기록 번호: ").strip()
    rows = run("SELECT * FROM study_logs WHERE id=%s", (log_id,), fetch=True)
    if rows:
        r = rows[0]
        print(f"\n--- 기록 #{r['id']} ---")
        print(f"제목: {r['title']}")
        print(f"날짜: {r['study_date']}")
        print(f"내용:\n{r['content']}\n")
    else:
        print("해당 기록이 없습니다.\n")

# 공부 기록 입력 (admin)
def add_log(user):
    title = input("제목: ").strip()
    content = input("내용: ").strip()
    date = input("날짜(YYYY-MM-DD, 비우면 오늘): ").strip()
    if date:
        run(
            "INSERT INTO study_logs (user_id, title, content, study_date) "
            "VALUES (%s, %s, %s, %s)",
            (user["id"], title, content, date),
        )
    else:
        run(
            "INSERT INTO study_logs (user_id, title, content) VALUES (%s, %s, %s)",
            (user["id"], title, content),
        )
    print("기록이 저장되었습니다.\n")

# 공부 기록 수정 (admin)
def edit_log():
    log_id = input("수정할 기록 번호: ").strip()
    title = input("새 제목: ").strip()
    content = input("새 내용: ").strip()
    run(
        "UPDATE study_logs SET title=%s, content=%s WHERE id=%s",
        (title, content, log_id),
    )
    print("기록이 수정되었습니다.\n")

# ===== 계정 관리 기능 (admin 전용) =====
# 계정 추가
def add_user():
    username = input("새 계정 아이디: ").strip()
    password = getpass.getpass("새 계정 비밀번호: ")
    role = input("역할(admin/guest): ").strip() or "guest"
    run(
        "INSERT INTO users (username, password, role) VALUES (%s, %s, %s)",
        (username, hash_pw(password), role),
    )
    print("계정이 추가되었습니다.\n")

# 비밀번호 변경
def change_pw():
    username = input("비밀번호 변경할 아이디: ").strip()
    password = getpass.getpass("새 비밀번호: ")
    run(
        "UPDATE users SET password=%s WHERE username=%s",
        (hash_pw(password), username),
    )
    print("비밀번호가 변경되었습니다.\n")

# 역할 변경
def change_role():
    username = input("역할 변경할 아이디: ").strip()
    role = input("새 역할(admin/guest): ").strip()
    run("UPDATE users SET role=%s WHERE username=%s", (role, username))
    print("역할이 변경되었습니다.\n")

# 계정 활성/비활성 전환
def toggle_active():
    username = input("활성/비활성 전환할 아이디: ").strip()
    run(
        "UPDATE users SET is_active = 1 - is_active WHERE username=%s",
        (username,),
    )
    print("계정 활성 상태가 전환되었습니다.\n")

# ===== 메뉴 =====
# admin 메뉴
def admin_menu(user):
    while True:
        print("===== ADMIN 메뉴 =====")
        print("1.기록 목록  2.기록 상세  3.기록 입력  4.기록 수정")
        print("5.계정 추가  6.비밀번호 변경  7.역할 변경  8.활성/비활성  0.로그아웃")
        sel = input("선택: ").strip()
        if sel == "1": list_logs()
        elif sel == "2": view_log()
        elif sel == "3": add_log(user)
        elif sel == "4": edit_log()
        elif sel == "5": add_user()
        elif sel == "6": change_pw()
        elif sel == "7": change_role()
        elif sel == "8": toggle_active()
        elif sel == "0": break
        else: print("잘못된 선택\n")

# guest 메뉴 (조회만)
def guest_menu(user):
    while True:
        print("===== GUEST 메뉴 =====")
        print("1.기록 목록  2.기록 상세  0.로그아웃")
        sel = input("선택: ").strip()
        if sel == "1": list_logs()
        elif sel == "2": view_log()
        elif sel == "0": break
        else: print("잘못된 선택\n")

# ===== 메인 =====
# 로그인 후 역할에 맞는 메뉴로 분기
def main():
    print("=== 공부 기록 관리 앱 ===\n")
    while True:
        user = login()
        if not user:
            if input("다시 시도? (y/n): ").strip().lower() != "y":
                break
            continue
        if user["role"] == "admin":
            admin_menu(user)
        else:
            guest_menu(user)
    print("프로그램을 종료합니다.")

if __name__ == "__main__":
    main()

    load_dotenv()가 .env 파일을 읽어 그 안의 DB_HOST=... 같은 값들을 환경변수로 만들어줍니다. 그러면 os.getenv("DB_HOST")로 꺼낼 수 있죠. os.getenv("DB_PORT", 3306)에서 두 번째 인자 3306은 값이 없을 때 쓰는 기본값입니다. 포트는 숫자여야 하므로 int()로 감쌌습니다. DB_CONFIG는 함수가 아니라 그 결과(딕셔너리)입니다. 한 번만 읽으면 되니까 맨 위에서 호출해 변수에 담아둡니다.

    def hash_pw(password):
    return hashlib.sha256(password.encode()).hexdigest()
    비밀번호를 그대로 저장하면 위험하니 해시로 변환합니다. .encode()로 문자열을 바이트로 바꾸고, sha256(...)으로 해시한 뒤 .hexdigest()로 사람이 읽을 수 있는 64자 16진수 문자열로 만듭니다. 같은 입력은 항상 같은 결과가 나오기 때문에, 로그인 시 입력한 비밀번호를 해시해서 DB에 저장된 해시와 비교하는 방식이 가능합니다.
    def run(sql, params=None, fetch=False):
    conn = get_conn()
    try:
        with conn.cursor(pymysql.cursors.DictCursor) as cur:
            cur.execute(sql, params or ())
            if fetch:
                return cur.fetchall()   # 조회면 결과 반환
            conn.commit()               # 변경이면 저장
    finally:
        conn.close()                    # 무조건 연결 닫기
    이게 핵심 헬퍼입니다. 모든 SQL이 이 함수 하나를 거칩니다. 짚어볼 점:
    • cur.execute(sql, params) — SQL 문자열에 값을 직접 끼워넣지 않고 %s 자리에 params를 따로 전달합니다. 이렇게 하면 SQL 인젝션 공격을 막을 수 있습니다. (params or ()는 params가 None이면 빈 튜플을 쓰라는 뜻)
    • DictCursor — 결과를 r['title']처럼 컬럼명으로 꺼낼 수 있게 해줍니다. (기본은 r[0] 같은 숫자 인덱스)
    • fetch 매개변수로 동작을 나눕니다. SELECT처럼 결과가 필요하면 fetch=True로 호출해 fetchall() 결과를 받고, INSERT/UPDATE는 fetch=False(기본)로 두고 commit()만 합니다.
    • try...finally — 도중에 에러가 나도 finally의 conn.close()는 반드시 실행돼서 연결이 새지 않습니다.

[예제 5-3] 고객 회사들이 주문한 주문건수를 주문건수가 많은 순서대로 보이시오. 이때 고객 회사의 정보로는 고객번호, 담당자명, 고객회사명을 보이시오.

from 고객 
		join 주문 
		on 고객..객번호  = 주문.고객번호 
group by 주문.고객번호 
order by count(*) desc;

[예제 5-4] 고객별(고객번호, 담당자명, 고객회사명)로 주문금액 합을 보이되, 주문금액 합이 많은 순서대로 보이시오.
select 고객.고객번호, 담당자명, 고객회사명, sum(단가 * 주문수량) as 주문금액합 
from 고객 
		join 주문 
		on 고객.고객번호 = 주문.고객번호 
				join 주문세부 
				on 주문.주문번호 = 주문세부.주문번호 
group by 고객.고객번호 
order by 주문금액합 desc;

[예제 5-6] 고객 테이블에서 담당자가 ‘이은광’인 경우의 고객번호, 고객회사명, 담당자명, 마일리지와 마일리지등급을 보이시오.

    def execute_sql(sql: str) -> tuple[list[str], list[tuple], Optional[str]]:
    sql = sql.strip()

    conn = None
    try:
        conn = mysql.connector.connect(**_DB_CONFIG, use_pure=True)
        cursor = conn.cursor()

        # 세미콜론으로 분리해서 하나씩 실행
        statements = [s.strip() for s in sql.split(';') if s.strip()]

        last_columns: list[str] = []
        last_rows: list[tuple] = []

        for stmt in statements:
            cursor.execute(stmt)
            if cursor.description:
                last_columns = [d[0] for d in cursor.description]
                last_rows = cursor.fetchall()
            else:
                last_columns = ['affected_rows']
                last_rows = [(cursor.rowcount,)]

        conn.commit()
        cursor.close()
        return last_columns, last_rows, None

    except Error as e:
        if conn:
            try:
                conn.rollback()
            except Exception:
                pass
        return [], [], str(e)
    finally:
        if conn and conn.is_connected():
            conn.close()
